یہ لیکچر سورۃ العادیات (سورہ نمبر ۱۰۰) کی تفسیر پر مبنی ہے، جس میں مجاہدین کے جنگی گھوڑوں کی قسم، انسان کی فطرت اور اس کے دو رخ، اور مال و جاہ کی محبت جیسے موضوعات پر روشنی ڈالی گئی ہے۔ بتایا گیا ہے کہ کس طرح انسان اپنی فطرت (جو روزِ ازل کے عہد کی گواہ ہے) کو دفن کر دیتا ہے اور نفسانی خواہشات کے تابع ہو کر خود اپنے خلاف گواہ بن جاتا ہے۔ اس کے ساتھ متعدد متعلقہ آیات کا حوالہ دیتے ہوئے انسانی زندگی، مال، اولاد اور دنیاوی وابستگیوں کا تجزیہ پیش کیا گیا ہے۔

مذکورہ آیات:

وَالْعَادِيَاتِ ضَبْحًا، وَالْمُورِيَاتِ قَدْحًا، فَالْمُغِيرَاتِ صُبْحًا، فَأَثَرْنَ بِهِ نَقْعًا
(Al-'Adiyati dabha, wal-Muriyati qadha, fal-Mughirati subha, fa-atharna bihi naq'a)
(قسم ہے ہانپتے ہوئے دوڑنے والے گھوڑوں کی، چنگاریاں نکالنے والوں کی، صبح کے وقت حملہ کرنے والوں کی، جنہوں نے اس سے گرد و غبار اڑا دیا)
سورۃ العادیات ۱۰۰: ۱-۴

وَالْخَيْلَ وَالْبِغَالَ وَالْحَمِيرَ لِتَرْكَبُوهَا
(Wal-khayla wal-bighala wal-hamira li-tarkabuha)
(اور گھوڑے، خچر اور گدھے پیدا کیے تاکہ تم ان پر سوار ہو)
سورۃ النحل ۱۶: ۸

وَلَكُمْ فِيهَا جَمَالٌ حِينَ تُرِيحُونَ وَحِينَ تَسْرَحُونَ
(Wa lakum fiha jamalun hina turihuna wa hina tasrahuna)
(اور تمہارے لیے ان میں زینت ہے جب تم انہیں شام کو واپس لاتے ہو اور جب صبح چراگاہ کو لے جاتے ہو)
سورۃ النحل ۱۶: ۶

لِيُظْهِرَهُ عَلَى الدِّينِ كُلِّهِ
(Li-yuzhirahu 'ala-d-dini kullih)
(تاکہ اسے تمام ادیان پر غالب کر دے)
سورۃ التوبہ ۹: ۳۳ (نیز سورۃ الفتح ۴۸: ۲۸ اور سورۃ الصف ۶۱: ۹ میں بھی مذکور)

إِنَّ الَّذِينَ آمَنُوا وَعَمِلُوا الصَّالِحَاتِ أُولَٰئِكَ هُمْ خَيْرُ الْبَرِيَّةِ
(Inna-lladhina amanu wa 'amilu-s-salihati ula'ika hum khayru-l-bariyya)
(بے شک جو لوگ ایمان لائے اور نیک عمل کیے، وہی بہترین مخلوق ہیں)
سورۃ البینہ ۹۸: ۷

وَإِذْ أَخَذَ رَبُّكَ مِنْ بَنِي آدَمَ مِنْ ظُهُورِهِمْ ذُرِّيَّتَهُمْ وَأَشْهَدَهُمْ عَلَىٰ أَنْفُسِهِمْ أَلَسْتُ بِرَبِّكُمْ قَالُوا بَلَىٰ
(A-lastu bi-Rabbikum, qalu bala)
(اور جب تیرے رب نے بنی آدم کی پشتوں سے ان کی نسل کو نکالا اور انہیں خود اپنے اوپر گواہ بنایا: کیا میں تمہارا رب نہیں ہوں؟ انہوں نے کہا: کیوں نہیں، ہم گواہ ہیں)
سورۃ الاعراف ۷: ۱۷۲

قَدْ خَابَ مَنْ دَسَّاهَا
(Qad khaba man dassaha)
(بے شک وہ ناکام ہوا جس نے اسے (نفس کو) دبا دیا)
سورۃ الشمس ۹۱: ۱۰

شَاهِدِينَ عَلَىٰ أَنْفُسِهِمْ بِالْكُفْرِ
(Shahidina 'ala anfusihim bi-l-kufr)
(اپنے آپ پر گواہی دیتے ہوئے کہ وہ کافر تھے)
سورۃ التوبہ ۹: ۱۷

إِنَّهُ لِرَبِّهِ لَكَنُودٌ
(Innahu li-Rabbihi la-kanud)
(بے شک وہ اپنے رب کا بڑا ناشکرا ہے)
سورۃ العادیات ۱۰۰: ۶

وَإِنَّهُ عَلَىٰ ذَٰلِكَ لَشَهِيدٌ
(Wa innahu 'ala dhalika la-shahid)
(اور وہ خود اس پر گواہ ہے)
سورۃ العادیات ۱۰۰: ۷

وَإِنَّهُ لِحُبِّ الْخَيْرِ لَشَدِيدٌ
(Wa innahu li-hubbi-l-khayri la-shadid)
(اور وہ مال کی محبت میں بہت سخت ہے)
سورۃ العادیات ۱۰۰: ۸

زُيِّنَ لِلنَّاسِ حُبُّ الشَّهَوَاتِ مِنَ النِّسَاءِ وَالْبَنِينَ وَالْقَنَاطِيرِ الْمُقَنْطَرَةِ مِنَ الذَّهَبِ وَالْفِضَّةِ وَالْخَيْلِ الْمُسَوَّمَةِ وَالْأَنْعَامِ وَالْحَرْثِ
(Zuyyina li-n-nasi hubbu-sh-shahawati mina-n-nisa'i wal-banina wal-qanatiri-l-muqantarati mina-dh-dhahabi wal-fiddati wal-khayli-l-musawwamati wal-an'ami wal-harth)
(لوگوں کے لیے خواہشات کی محبت خوشنما بنا دی گئی ہے: عورتوں اور بیٹوں کی، سونے چاندی کے جمع کردہ ڈھیروں کی، نشان دار گھوڑوں، مویشیوں اور کھیتی کی)
سورۃ آل عمران ۳: ۱۴

قُلْ إِنْ كُنْتُمْ تُحِبُّونَ اللهَ فَاتَّبِعُونِي يُحْبِبْكُمُ اللهُ
(Qul in kuntum tuhibbuna-Llaha fattabi'uni yuhbibkumu-Llah)
(کہہ دیجیے: اگر تم اللہ سے محبت رکھتے ہو تو میری پیروی کرو، اللہ تم سے محبت کرے گا)
سورۃ آل عمران ۳: ۳۱

إِنَّ الْمُلُوكَ إِذَا دَخَلُوا قَرْيَةً أَفْسَدُوهَا وَجَعَلُوا أَعِزَّةَ أَهْلِهَا أَذِلَّةً
(Inna-l-muluka idha dakhalu qaryatan afsaduha wa ja'alu a'izzata ahliha adhillah)
(بے شک بادشاہ جب کسی بستی میں داخل ہوتے ہیں تو اسے تباہ کر دیتے ہیں اور اس کے معزز لوگوں کو ذلیل کر دیتے ہیں)
سورۃ النمل ۲۷: ۳۴

تَشْهَدُ عَلَيْهِمْ أَلْسِنَتُهُمْ بِمَا كَانُوا يَعْمَلُونَ
(Tashhadu 'alayhim alsinatuhum bima kanu ya'maluna)
(ان کی زبانیں انہی کے خلاف اس پر گواہی دیں گی جو وہ کرتے تھے)
سورۃ النور ۲۴: ۲۴

مِنْ صَلْصَالٍ مِنْ حَمَإٍ مَسْنُونٍ
(Min salsalin min hama'in masnun)
(بجتی ہوئی مٹی سے، جو سڑے ہوئے گارے سے بنی تھی)
سورۃ الحجر ۱۵: ۲۶

مَالِ هَٰذَا الْكِتَابِ لَا يُغَادِرُ صَغِيرَةً وَلَا كَبِيرَةً إِلَّا أَحْصَاهَا
(Ma li-hadha-l-kitabi la yughadiru saghiratan wa la kabiratan illa ahsaha)
(اس کتاب کو کیا ہوا کہ نہ چھوٹی نہ بڑی بات کو چھوڑا، سب کو شمار کر رکھا ہے)
سورۃ الکہف ۱۸: ۴۹

عَلِمَتْ نَفْسٌ مَا أَحْضَرَتْ
('Alimat nafsun ma ahdarat)
(ہر نفس جان لے گا جو کچھ اس نے حاضر کیا)
سورۃ التکویر ۸۱: ۱۴

كِرَامًا كَاتِبِينَ
(Kiraman katibin)
(معزز لکھنے والے (فرشتے))
سورۃ الانفطار ۸۲: ۱۱

تفسیری خلاصہ:

یہ سورت مدینہ میں نازل ہوئی اور دفاعِ اسلام میں شریک ان جنگی گھوڑوں کی قسم سے شروع ہوتی ہے جو ہانپتے ہوئے دوڑتے، سموں سے چنگاریاں نکالتے اور صبح کے وقت دشمن پر یکبارگی حملہ آور ہوتے تھے۔ یہ قسم دراصل مجاہدین کے جہاد کی طرف اشارہ ہے، اور یہی جہاد تھا جس نے اسلام کو قائم اور غالب کیا، جیسا کہ فرمایا گیا کہ اللہ اپنے دین کو تمام ادیان پر غالب کرے گا۔ جو مومن جہاد اور عملِ صالح سے آراستہ ہوں وہی بہترین مخلوق ہیں، اور بریت کا حقیقی مفہوم صرف نیک اعمال نہیں بلکہ وہ کھلا دل ہے جو تمام انسانیت کے لیے دعا اور فکرمند رہتا ہے۔

اس کے بعد قرآن انسان کو پچاس سے زائد مقامات پر تنبیہ کرتا ہے، اور اس کی بنیاد اس عہدِ ازل سے جوڑتا ہے جب روحِ انسانی نے اپنے رب کی ربوبیت کی گواہی دی تھی۔ یہ فطرت کبھی مٹتی نہیں، مگر انسان جب اپنی خواہشات اور نفسانی رغبتوں میں غرق ہو کر اسے دبا دیتا اور خاموش کر دیتا ہے تو وہی فطرت روزِ قیامت اس کے خلاف گواہ بن جاتی ہے۔ انسان اپنے آپ پر خود گواہ ہوگا، اور اس کے اعضا (زبان، ہاتھ وغیرہ) بھی اس کے اعمال کی گواہی دیں گے۔ یہ گواہی کوئی اقرار نہیں بلکہ اس باطنی و ظاہری فطرت کی گواہی ہے جو انسان کے نفسِ حیوانی کے خلاف کھڑی ہوتی ہے۔

انسان کی سب سے بڑی کمزوری مال، اولاد، جاہ اور زمین کی چار قسم کی وابستگیوں سے محبت ہے: معدنیات (سونا چاندی)، حیوانات (نشان دار گھوڑے اور مویشی)، نباتات (کھیتی) اور انسان (عورتیں اور بیٹے)۔ اللہ نے انہیں خوشنما بنا دیا ہے تاکہ انسان کا امتحان ہو، مگر حقیقی کامیابی یہ ہے کہ انسان اپنی محبت کو ان چار چیزوں سے ہٹا کر اللہ کی محبت میں مرکوز کر دے، جیسا کہ اہلِ اللہ اور شہدائے کرام نے اپنی جانیں اور مال اللہ کی راہ میں قربان کر دیے۔ اصل دعوت یہی ہے کہ انسان اپنی دبی ہوئی فطرت کو زندہ کرے، اپنے دل کے پردے اٹھائے، اور روزِ قیامت سے پہلے ہی اپنے آپ کا محاسبہ کر کے بہترین معاشرے اور بہترین انسان کی تشکیل میں شامل ہو جائے۔
